<script>
/* Departure data for the journey cards below. The arrays here are the
   instant-paint FALLBACK — the LIVE Worker feed loads immediately and can
   only RAISE a date's status (available - nearing - soldout); it never
   removes dates and never blanks the list, so a stale or unreachable feed
   cannot break this page. Past departures are KEPT and shown sold out.
   To add a new season: edit RAW here, the Worker SCHEDULE, and Checkfront.
   Never edit this when a date sells out — that happens automatically. */
(function(){
  var RAW = {
    'winter-2026'  : ['2026-06-27','2026-07-18','2026-08-01','2026-09-05','2026-09-12'],
    'pinnacle-2027': ['2027-01-14'],
    '11day-2627'   : ['2026-11-02','2026-11-30','2027-02-01','2027-02-22','2027-03-01','2027-03-29'],
    '14day-2627'   : ['2026-11-09','2026-11-16','2026-11-23','2027-01-04','2027-01-11','2027-01-18','2027-01-25','2027-02-08','2027-02-15','2027-03-08','2027-03-15','2027-03-22','2027-04-05','2027-04-12'],
    'winter-2027'  : ['2027-07-10','2027-07-24','2027-08-07','2027-08-14','2027-08-21','2027-08-28'],
    '11day-2728'   : ['2027-09-27','2027-10-25','2027-11-29','2028-01-10','2028-02-28','2028-03-06','2028-04-17'],
    '14day-2728'   : ['2027-10-04','2027-10-11','2027-10-18','2027-11-01','2027-11-08','2027-11-15','2027-11-22','2027-12-06','2028-01-17','2028-01-24','2028-01-31','2028-02-07','2028-02-14']
  };
  var SOLDOUT = {
    'winter-2026': ['2026-06-27','2026-07-18']
  };
  var ADD = { 'winter-2026':11,'winter-2027':11,'pinnacle-2027':15/*15 nights - ends 29 Jan 2027; not the usual days-1*/,'11day-2627':10,'11day-2728':10,'14day-2627':13,'14day-2728':13 };
  function endOf(iso,n){ var d=new Date(iso+'T00:00:00Z'); d.setUTCDate(d.getUTCDate()+n); return d.toISOString().slice(0,10); }
  var today=new Date().toISOString().slice(0,10), data={};
  Object.keys(RAW).forEach(function(k){
    var sold=SOLDOUT[k]||[];
    var all=RAW[k].map(function(x){
      var st='available';
      if(x < today){ st='soldout'; }                 /* departed - kept, shown sold out */
      else if(sold.indexOf(x) >= 0){ st='soldout'; } /* manually marked sold out */
      return {start:x,end:endOf(x,ADD[k]),status:st};
    });
    var nx=null;
    for(var i=0;i < all.length;i++){ if(all[i].status!=='soldout'){ nx=all[i].start; break; } }
    data[k]={ all:all, next:nx };
  });
  window.__seheDepartureData=data;   /* instant-paint fallback */

  /* ---- Live availability feed (shared, guarded) ---------------------------
     One fetch of the Worker per page load. ESCALATE-ONLY MERGE: keep our own
     baked date list, never replace it; only raise a date's status, matched by
     start date. An empty, stale or partial response cannot drop dates or keys.
     If the feed is unreachable the fallback remains — the page never breaks. */
  var WORKER_URL='https://sehe-next-departures.ben-757.workers.dev/';
  window.__seheDeparturePromise=null;
  window.__seheLoadDepartures=function(){
    if(window.__seheDeparturePromise) return window.__seheDeparturePromise;
    window.__seheDeparturePromise=fetch(WORKER_URL,{cache:'no-store'})
      .then(function(r){ if(!r.ok) throw new Error('Worker '+r.status); return r.json(); })
      .then(function(live){
        if(live && typeof live==='object' && !Array.isArray(live)){
          var RANK={available:0,nearing:1,soldout:2}, cur=window.__seheDepartureData||{};
          Object.keys(cur).forEach(function(k){
            var lv=live[k]; if(!lv||!lv.all||!lv.all.length||!cur[k]||!cur[k].all) return;
            var lmap={}; lv.all.forEach(function(x){ if(x&&x.start) lmap[x.start]=x.status||'available'; });
            cur[k].all.forEach(function(d){ var ls=lmap[d.start]; if(ls && (RANK[ls]||0)>(RANK[d.status]||0)) d.status=ls; });
            var nx=null; for(var i=0;i < cur[k].all.length;i++){ if(cur[k].all[i].status!=='soldout'){ nx=cur[k].all[i].start; break; } }
            cur[k].next=nx;
          });
        }
        return window.__seheDepartureData;
      })
      .catch(function(err){ if(window.console&&console.warn) console.warn('SEHE departures: live feed unavailable, using fallback list.',err); return window.__seheDepartureData; });
    return window.__seheDeparturePromise;
  };
  window.__seheLoadDepartures();   /* kick off early so it is in flight ASAP */
})();
</script>

(function() {
  var MONTHS_SHORT = ['Jan','Feb','Mar','Apr','May','Jun','Jul','Aug','Sep','Oct','Nov','Dec'];
  function formatDate(isoStr) { var parts = isoStr.split('-'); var m = parseInt(parts[1], 10), d = parseInt(parts[2], 10); return d + ' ' + MONTHS_SHORT[m - 1] + ' ' + parts[0]; }
  function availOf(info) { return (info && info.all ? info.all : []).filter(function(x) { return x && x.status !== 'soldout'; }); }
  function nextAvail(info) { if (info && info.next) return info.next; var a = availOf(info); return a.length ? a[0].start : null; }
  function updateDepartureCount(card, info) {
    var valueEl = card.querySelector('[data-departure-value]');
    var labelEl = card.querySelector('[data-departure-label]');
    if (!valueEl || !labelEl) return;
    var all = availOf(info);
    if (all.length === 0) { valueEl.textContent = '—'; labelEl.textContent = 'Departures'; return; }
    if (all.length === 1) { var parts = all[0].start.split('-'); var m = parseInt(parts[1], 10), d = parseInt(parts[2], 10); valueEl.textContent = d + ' ' + MONTHS_SHORT[m - 1]; labelEl.textContent = 'Departs ' + parts[0]; return; }
    valueEl.textContent = all.length; labelEl.textContent = 'Departures';
  }
  function renderCard(card, data) {
    var tourId = card.getAttribute('data-tour');
    if (!tourId) return;
    var info = data[tourId];
    if (!info || !info.all || !info.all.length) return;   /* data not ready / no dates — keep the baked card, never wipe it */
    var placeholder = card.querySelector('[data-next-placeholder]');
    if (placeholder) {
      var nx = nextAvail(info);
      if (!nx) {
        placeholder.classList.add('journey-card-next-soldout');
        placeholder.querySelector('.journey-card-next-label').textContent = 'Season Complete';
        placeholder.querySelector('.journey-card-next-date').textContent = 'New season opening soon';
      } else {
        placeholder.classList.remove('journey-card-next-soldout');
        placeholder.querySelector('.journey-card-next-label').textContent = 'Next Departure';
        placeholder.querySelector('.journey-card-next-date').textContent = formatDate(nx);
      }
    }
    updateDepartureCount(card, info);
  }
  var widget = document.querySelector('.sehe-journeys');
  if (!widget) return;
  function renderAll() { var data = window.__seheDepartureData || {}; widget.querySelectorAll('.journey-card[data-tour]').forEach(function(card) { renderCard(card, data); }); }
  renderAll();   /* instant paint from fallback / already-loaded live data */
  if (typeof window.__seheLoadDepartures === 'function') { window.__seheLoadDepartures().then(renderAll); }   /* re-render with live */
})();

<script>
(function() {
  /* Shared video modal — binds EVERY [data-video-frame] on the page (the film
     section and the traveller-stories video) to the one modal. Enhance-only:
     missing elements mean it simply does nothing. */
  if (!document.querySelectorAll || !document.getElementById) return;
  var modal = document.getElementById('seheVideoModal');
  var content = document.getElementById('seheVideoModalContent');
  var closeBtn = document.getElementById('seheVideoClose');
  var frames = document.querySelectorAll('[data-video-frame]');
  if (!modal || !content || !closeBtn || frames.length === 0) return;
  function clearPlayers() {
    var existing = content.querySelectorAll('video, iframe');
    existing.forEach(function(el) { el.remove(); });
  }
  function openModal(frame) {
    var src = frame.getAttribute('data-video-src');
    var type = (frame.getAttribute('data-video-type') || 'mp4').toLowerCase();
    if (!src) return;
    clearPlayers();
    if (type === 'vimeo') {
      var vframe = document.createElement('iframe');
      vframe.src = 'https://player.vimeo.com/video/' + src + '?autoplay=1';
      vframe.allow = 'autoplay; fullscreen; picture-in-picture';
      vframe.allowFullscreen = true;
      content.appendChild(vframe);
    } else if (type === 'youtube') {
      var yframe = document.createElement('iframe');
      yframe.src = 'https://www.youtube.com/embed/' + src + '?autoplay=1&rel=0';
      yframe.allow = 'autoplay; encrypted-media; picture-in-picture; fullscreen';
      yframe.allowFullscreen = true;
      content.appendChild(yframe);
    } else {
      var video = document.createElement('video');
      video.src = src; video.controls = true; video.autoplay = true; video.playsInline = true;
      content.appendChild(video);
    }
    modal.classList.add('is-open');
    document.body.style.overflow = 'hidden';
  }
  function closeModal() {
    modal.classList.remove('is-open');
    document.body.style.overflow = '';
    var v = content.querySelector('video');
    if (v) { try { v.pause(); } catch(e) {} }
    clearPlayers();
  }
  frames.forEach(function(frame) {
    frame.addEventListener('click', function(e) { if (e && e.preventDefault) e.preventDefault(); openModal(frame); });
    frame.addEventListener('keydown', function(e) { if (e.key === 'Enter' || e.key === ' ') { e.preventDefault(); openModal(frame); } });
  });
  closeBtn.addEventListener('click', closeModal);
  modal.addEventListener('click', function(e) { if (e.target === modal) closeModal(); });
  document.addEventListener('keydown', function(e) { if (e.key === 'Escape' && modal.classList.contains('is-open')) closeModal(); });
})();
</script>
