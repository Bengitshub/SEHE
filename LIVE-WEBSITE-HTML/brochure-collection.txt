(function(){
  /* Departure dates for the cards — Kirsty's full schedule per tour. Every date
     shows; dates listed in SOLDOUT (not bookable in Checkfront) render RED.
     The live feed below only ESCALATES a date's colour (available -> filling
     fast -> sold) and never reverts this list. To change departures by hand,
     edit RAW / SOLDOUT here. */
  var RAW = {
    'winter-2026'  : ['2026-06-27','2026-07-18','2026-08-01','2026-09-05','2026-09-12'],
    'pinnacle-2027': ['2027-01-14'],
    '11day-2627'   : ['2026-10-05','2026-11-02','2026-11-30','2027-02-01','2027-02-22','2027-03-01','2027-03-29'],
    '14day-2627'   : ['2026-10-12','2026-10-19','2026-10-26','2026-11-09','2026-11-16','2026-11-23','2027-01-04','2027-01-11','2027-01-18','2027-01-25','2027-02-08','2027-02-15','2027-03-08','2027-03-15','2027-03-22','2027-04-05','2027-04-12'],
    'winter-2027'  : ['2027-07-10','2027-07-24','2027-08-07','2027-08-14','2027-08-21','2027-08-28'],
    '11day-2728'   : ['2027-09-27','2027-10-25','2027-11-29','2028-01-10','2028-02-28','2028-03-06','2028-04-17'],
    '14day-2728'   : ['2027-10-04','2027-10-11','2027-10-18','2027-11-01','2027-11-08','2027-11-15','2027-11-22','2027-12-06','2028-01-17','2028-01-24','2028-01-31','2028-02-07','2028-02-14']
  };
  var SOLDOUT = {
    'winter-2026': ['2026-06-27','2026-07-18','2026-08-01','2026-09-05','2026-09-12'],
    '11day-2627' : ['2026-10-05'],
    '14day-2627' : ['2026-10-12','2026-10-19','2026-10-26']
  };
  var ADD = { 'winter-2026':11,'winter-2027':11,'pinnacle-2027':14,'11day-2627':10,'11day-2728':10,'14day-2627':13,'14day-2728':13 };
  function endOf(iso,n){ var d=new Date(iso+'T00:00:00Z'); d.setUTCDate(d.getUTCDate()+n); return d.toISOString().slice(0,10); }
  var today=new Date().toISOString().slice(0,10), data={};
  Object.keys(RAW).forEach(function(k){
    var sold=SOLDOUT[k]||[];
    var all=RAW[k].filter(function(x){return x>=today;}).map(function(x){return {start:x,end:endOf(x,ADD[k]),status:sold.indexOf(x)>=0?'soldout':'available'};});
    var av=all.filter(function(x){return x.status!=='soldout';});
    data[k]={ all:all, next: av.length?av[0].start:null };
  });
  window.__seheDepartureData=data;
  /* ---- LIVE DATES (the worker is the source of truth) --------------------
     The worker holds the full schedule and reads Checkfront for what's available
     vs sold out. We OVERRIDE the baked list per tour with the worker's live list,
     so sell-outs, freed cancellations and twin-share dates all update on their
     own — no re-editing. SAFETY: if the worker is unreachable, or returns nothing
     for a tour, we keep the baked list (we never wipe it to empty). */
  var WORKER_URL='https://sehe-next-departures.ben-757.workers.dev/';
  window.__seheDeparturePromise=null;
  window.__seheLoadDepartures=function(){
    if(window.__seheDeparturePromise) return window.__seheDeparturePromise;
    window.__seheDeparturePromise=fetch(WORKER_URL,{cache:'no-store'})
      .then(function(r){ if(!r.ok) throw new Error('Worker '+r.status); return r.json(); })
      .then(function(live){
        if(live && typeof live==='object' && !Array.isArray(live)){
          Object.keys(window.__seheDepartureData).forEach(function(k){
            var lt=live[k];
            if(!lt || !lt.all || !lt.all.length) return;   /* nothing live for this tour -> keep the baked list (never wipe) */
            var all=lt.all.filter(function(d){ return d && d.start; }).map(function(d){
              return { start:d.start, end:d.end||endOf(d.start,ADD[k]), status:d.status||'available' };
            });
            if(!all.length) return;
            window.__seheDepartureData[k].all=all;                       /* worker wins */
            var av=all.filter(function(d){ return d.status!=='soldout'; });
            window.__seheDepartureData[k].next= lt.next || (av.length?av[0].start:null);
          });
        }
        return window.__seheDepartureData;
      })
      .catch(function(err){ if(window.console&&console.warn) console.warn('SEHE departures: live feed unavailable, keeping the in-page list.',err); return window.__seheDepartureData; });
    return window.__seheDeparturePromise;
  };
})();

 (function() {
  // ──────────────────────────────────────────────────────────────────
  // SEHE BROCHURE GRID — DEPARTURE DATES + LIVE COUNT (never-wipe render)
  // Reads window.__seheDepartureData (set by the shared block at the top of
  // this widget — the SAME master list the tour pages and Journeys index use).
  // Per card it ENHANCES the baked HTML only: it re-renders the month-grouped
  // date list and the "Departures" count, then applies the escalate-only live
  // status. If a tour's data is missing it RETURNS and leaves the baked HTML
  // untouched — so a render pass before the data block, or an offline Worker,
  // can never blank or revert the dates.
  // ──────────────────────────────────────────────────────────────────
  var MONTHS_SHORT = ['Jan','Feb','Mar','Apr','May','Jun','Jul','Aug','Sep','Oct','Nov','Dec'];
  function availOf(info) { return (info && info.all ? info.all : []).filter(function(x) { return x && x.status !== 'soldout'; }); }
  function updateCount(card, info) {
    var valueEl = card.querySelector('[data-departure-value]');
    var labelEl = card.querySelector('[data-departure-label]');
    if (!valueEl || !labelEl) return;
    var av = availOf(info);
    if (av.length === 0) { valueEl.textContent = '—'; labelEl.textContent = 'Departures'; return; }
    if (av.length === 1) {
      var p = av[0].start.split('-'), m = parseInt(p[1], 10), d = parseInt(p[2], 10);
      valueEl.textContent = d + ' ' + MONTHS_SHORT[m - 1];
      labelEl.textContent = 'Departs ' + p[0];
      return;
    }
    valueEl.textContent = av.length;
    labelEl.textContent = 'Departures';
  }
  function buildDatesHtml(all) {
    if (!all || !all.length) return '';
    var groups = {}, order = [], sold = 0, near = 0;
    for (var i = 0; i < all.length; i++) {
      var dep = all[i], p = dep.start.split('-'), m = parseInt(p[1], 10);
      var key = p[0] + '-' + p[1];
      if (!groups[key]) { groups[key] = { label: MONTHS_SHORT[m - 1] + ' ' + p[0], days: [] }; order.push(key); }
      if (dep.status === 'soldout') sold++; else if (dep.status === 'nearing') near++;
      groups[key].days.push({ d: parseInt(p[2], 10), status: dep.status });
    }
    var rows = '';
    for (var o = 0; o < order.length; o++) {
      var g = groups[order[o]];
      var days = g.days.map(function(day) {
        if (day.status === 'soldout') return '<span class="jc-day-sold" title="Sold out">' + day.d + '</span>';
        if (day.status === 'nearing') return '<span class="jc-day-nearing" title="Filling fast">' + day.d + '</span>';
        return '<span>' + day.d + '</span>';
      }).join('<span class="jc-day-sep">·</span>');
      rows += '<div class="jc-month"><span class="jc-month-label">' + g.label + '</span><span class="jc-days">' + days + '</span></div>';
    }
    var legend = '';
    if (sold > 0 || near > 0) {
      legend = '<div class="jc-legend"><span class="jc-legend-item"><span class="jc-legend-dot jc-legend-avail"></span>Available</span>';
      if (near > 0) legend += '<span class="jc-legend-item"><span class="jc-legend-dot jc-legend-near"></span>Filling fast</span>';
      if (sold > 0) legend += '<span class="jc-legend-item"><span class="jc-legend-dot jc-legend-sold"></span>Sold out</span>';
      legend += '</div>';
    }
    return '<span class="journey-card-dates-label">All departure dates</span><div class="jc-months">' + rows + '</div>' + legend;
  }
  function renderDatesList(card, info) {
    if (!info || !info.all || !info.all.length) return;   // no data -> keep the baked-in list (never wipe it)
    var html = buildDatesHtml(info.all);
    if (!html) return;
    var existing = card.querySelector('.journey-card-departures');
    if (existing) { existing.innerHTML = html; return; }
    var el = document.createElement('div');
    el.className = 'journey-card-departures';
    el.innerHTML = html;
    var anchor = card.querySelector('.tour-card-facts');
    if (anchor && anchor.parentNode) { anchor.parentNode.insertBefore(el, anchor.nextSibling); }
    else { var body = card.querySelector('.tour-card-body'); if (body) body.appendChild(el); }
  }
  function renderCard(card, data) {
    var key = card.getAttribute('data-tour-key');
    if (!key) return;
    var info = data[key];
    if (!info || !info.all || !info.all.length) return;   // no data yet -> leave the card's baked content untouched
    updateCount(card, info);
    renderDatesList(card, info);
  }
  var widget = document.querySelector('.sehe-brochures');
  if (!widget) return;
  function renderAll() { var data = window.__seheDepartureData || {}; widget.querySelectorAll('.tour-card[data-tour-key]').forEach(function(card) { renderCard(card, data); }); }
  renderAll();   /* paint the in-page list immediately */
  if (typeof window.__seheLoadDepartures === 'function') { window.__seheLoadDepartures().then(renderAll); }   /* then apply live status (escalate-only merge) */
})();
