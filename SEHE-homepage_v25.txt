<!-- ===== LIVE NEXT-DEPARTURE CHIPS + FILM REVEAL (enhance-only) ===== -->
<script>
  (function() {
  /* Journey cards: refresh each "Next departure" chip from the live
     availability worker. ENHANCE-ONLY: chips ship with baked dates; a chip
     is only overwritten when the feed returns a usable next date for that
     tour. No chip is ever emptied. */
  var MONTHS = ['January','February','March','April','May','June','July','August','September','October','November','December'];
  function pretty(iso) {
    var p = iso.split('-');
    if (p.length !== 3) return null;
    var m = parseInt(p[1], 10);
    if (!(m >= 1 && m <= 12)) return null;
    return parseInt(p[2], 10) + ' ' + MONTHS[m - 1] + ' ' + p[0];
  }
  function apply(data) {
    if (!data || typeof data !== 'object') return;
    var chips = document.querySelectorAll('[data-shx-next]');
    for (var i = 0; i < chips.length; i += 1) {
      var key = chips[i].getAttribute('data-shx-next');
      var tour = data[key];
      if (!tour || !tour.next) continue;
      var label = pretty(tour.next);
      if (label) { chips[i].textContent = label; }
    }
  }
  function go() {
    try {
      fetch('https://sehe-next-departures.ben-757.workers.dev/', { cache: 'no-store' })
        .then(function(r) { if (!r.ok) { throw new Error('feed ' + r.status); } return r.json(); })
        .then(apply)
        .catch(function() { /* keep baked dates */ });
    } catch (err) { /* keep baked dates */ }
  }
  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', go);
  } else {
    go();
  }
  /* FILM section: the video layer stays invisible until the Vimeo player
     proves it is alive (any postMessage from vimeo). If the player never
     loads, the poster photo simply remains. Photo is the base state. */
  window.addEventListener('message', function(ev) {
    if (ev && typeof ev.origin === 'string' && ev.origin.indexOf('vimeo.com') !== -1) {
      var media = document.querySelector('.shx-film-media');
      if (media && media.className.indexOf('is-live') === -1) { media.className += ' is-live'; }
    }
  });
})();
</script>
