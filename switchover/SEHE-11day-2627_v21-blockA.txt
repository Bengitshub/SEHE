<script>
(function(){
  var RAW = {
    'winter-2026'  : ['2026-06-27','2026-07-18','2026-08-01','2026-09-05','2026-09-12'],
    'pinnacle-2027': ['2027-01-14'],
    '11day-2627'   : ['2026-10-05','2026-11-02','2026-11-30','2027-02-01','2027-02-22','2027-03-01','2027-03-29'],
    '14day-2627'   : ['2026-10-12','2026-10-19','2026-10-26','2026-11-09','2026-11-16','2026-11-23','2027-01-04','2027-01-11','2027-01-18','2027-01-25','2027-02-08','2027-02-15','2027-03-08','2027-03-15','2027-03-22','2027-04-05','2027-04-12'],
    'winter-2027'  : ['2027-07-10','2027-07-24','2027-08-07','2027-08-14','2027-08-21','2027-08-28'],
    '11day-2728'   : ['2027-09-27','2027-10-25','2027-11-29','2028-01-10','2028-02-28','2028-03-06','2028-04-17'],
    '14day-2728'   : ['2027-10-04','2027-10-11','2027-10-18','2027-11-01','2027-11-08','2027-11-15','2027-11-22','2027-12-06','2028-01-17','2028-01-24','2028-01-31','2028-02-07','2028-02-14']
  };
  var SOLDOUT = {
    'winter-2026': ['2026-06-27','2026-07-18','2026-08-01','2026-09-05'],
    '11day-2627' : ['2026-10-05','2026-11-02'],
    '14day-2627' : ['2026-10-12','2026-10-19','2026-10-26']
  };
  var ADD = { 'winter-2026':11,'winter-2027':11,'pinnacle-2027':15,'11day-2627':10,'11day-2728':10,'14day-2627':13,'14day-2728':13 };
  function endOf(iso,n){ var d=new Date(iso+'T00:00:00Z'); d.setUTCDate(d.getUTCDate()+n); return d.toISOString().slice(0,10); }
  var today=new Date().toISOString().slice(0,10), data={};
  Object.keys(RAW).forEach(function(k){
    var sold=SOLDOUT[k]||[];
    var all=RAW[k].filter(function(x){return x>=today;}).map(function(x){return {start:x,end:endOf(x,ADD[k]),status:sold.indexOf(x)>=0?'soldout':'available'};});
    var av=all.filter(function(x){return x.status!=='soldout';});
    data[k]={ all:all, next: av.length?av[0].start:null };
  });
  window.__seheDepartureData=data;

  /* ---- LIVE DATES (the worker is the source of truth) --------------------
     The worker holds the full schedule and reads Checkfront for what's available
     vs sold out. We OVERRIDE the baked list per tour with the worker's live list,
     so sell-outs, freed cancellations and twin-share dates all update on their
     own — no re-pasting. SAFETY: if the worker is unreachable, or returns nothing
     for a tour, we keep the baked list (we never wipe it to empty). */
  var WORKER_URL='https://sehe-next-departures.ben-757.workers.dev/';
  window.__seheDeparturePromise=null;
  window.__seheLoadDepartures=function(){
    if(window.__seheDeparturePromise) return window.__seheDeparturePromise;
    window.__seheDeparturePromise=fetch(WORKER_URL,{cache:'no-store'})
      .then(function(r){ if(!r.ok) throw new Error('Worker '+r.status); return r.json(); })
      .then(function(live){
        if(live && typeof live==='object' && !Array.isArray(live)){
          Object.keys(window.__seheDepartureData).forEach(function(k){
            var lt=live[k];
            if(!lt || !lt.all || !lt.all.length) return;   /* nothing live for this tour -> keep the baked list (never wipe) */
            var all=lt.all.filter(function(d){ return d && d.start; }).map(function(d){
              return { start:d.start, end:d.end||endOf(d.start,ADD[k]), status:d.status||'available' };
            });
            if(!all.length) return;
            window.__seheDepartureData[k].all=all;                       /* worker wins */
            var av=all.filter(function(d){ return d.status!=='soldout'; });
            window.__seheDepartureData[k].next= lt.next || (av.length?av[0].start:null);
          });
        }
        return window.__seheDepartureData;
      })
      .catch(function(err){ if(window.console&&console.warn) console.warn('SEHE departures: live feed unavailable, keeping the in-page list.',err); return window.__seheDepartureData; });
    return window.__seheDeparturePromise;
  };
})();
</script>

  (function() {
  /* Brochure form DOCKING (switch-over sandwich pages). The real Duda form
     widget is pasted below this HTML block so its webhook keeps working; on
     the published site this script moves that widget INTO the brochure card
     form slot as soon as it appears in the DOM, so the form renders exactly
     where the design intends. Enhance-only: in the Duda editor, on this
     master preview, or if no Duda form widget exists on the page, nothing is
     touched and the layout stays as pasted. */
  if (!/(^|\.)siredmundhillaryexplorer\.com$/.test(window.location.hostname)) return;
  var slot = document.querySelector('[data-sehe-form-slot]');
  if (!slot) return;
  var tries = 0;
  var timer = window.setInterval(function() {
    tries += 1;
    if (tries > 75) { window.clearInterval(timer); return; }
    var all = document.querySelectorAll('div[data-element-type="dContactUsRespId"]');
    var w = null;
    for (var i = 0; i < all.length; i += 1) {
      if (all[i].querySelector('input[name="dmform-1"]') && all[i].querySelector('input[name="dmform-3"]')) { w = all[i]; break; }
    }
    if (!w) return;
    window.clearInterval(timer);
    var oldParent = w.parentElement;
    slot.appendChild(w);
    slot.removeAttribute('hidden');
    /* The widget carries geometry from its old home: Duda instance CSS and the
       site theme still size/paint its internal wrappers (fixed button width, a
       painted submit CONTAINER that shows as a slab behind the button, left
       padding that mis-aligns the fields). Class-level CSS cannot outrank
       instance rules, so flatten the geometry INLINE at dock time — inline
       importants beat every stylesheet. Colours/typography stay with the skin. */
    var flat = function(el, geom) {
      if (!el || !el.style) return;
      for (var k in geom) { if (geom.hasOwnProperty(k)) { el.style.setProperty(k, geom[k], 'important'); } }
    };
    flat(w, { padding: '0', margin: '0', width: '100%', 'max-width': 'none', background: 'transparent', 'background-color': 'transparent', 'min-height': '0', height: 'auto' });
    flat(w.querySelector('.dmform-wrapper'), { padding: '0', margin: '0', width: '100%', 'max-width': 'none', background: 'transparent' });
    flat(w.querySelector('form'), { padding: '0', margin: '0', width: '100%', 'max-width': 'none' });
    flat(w.querySelector('.spacing-container'), { padding: '0', margin: '0', width: '100%' });
    var ttl = w.querySelector('.dmform-title');
    if (ttl) { ttl.style.setProperty('display', 'none', 'important'); }
    var rows = w.querySelectorAll('.dmforminput');
    for (var j = 0; j < rows.length; j += 1) {
      flat(rows[j], { float: 'none', width: '100%', margin: '0 0 16px 0', padding: '0', 'min-height': '0', height: 'auto', background: 'transparent' });
    }
    /* the fields themselves: Duda's device stylesheets can size inputs past
       their column (seen on iPhone) — inline width pins the fields and the
       button to the identical box so they are always edge-flush */
    var flds = w.querySelectorAll('input[type="text"], input[type="email"], select');
    for (var q = 0; q < flds.length; q += 1) {
      flat(flds[q], { display: 'block', width: '100%', 'max-width': '100%', 'box-sizing': 'border-box', margin: '0' });
    }
    var sub = w.querySelector('.dmformsubmit');
    flat(sub, { float: 'none', display: 'block', width: '100%', 'max-width': '100%', margin: '22px 0 0 0', padding: '0', background: 'transparent', 'background-color': 'transparent', border: '0', 'border-radius': '0', 'box-shadow': 'none', 'min-height': '0', height: 'auto', 'text-align': 'left' });

    /* the submit is an INPUT and inputs cannot wrap their label, so on narrow
       screens the long label overflows — step the type down responsively.
       Tuned after Ben's phone review: 12.5px covers all normal phones (the
       old 360px step caught most Androids at 11px — too small); 11px only on
       tiny (340px and under) screens; text-indent equal to the letter-spacing
       cancels the trailing-space bias so the label sits optically centre. */
    var sizeBtn = function() {
      if (!btn) return;
      var fs = '13.5px', ls = 1.4, pad = '17px 24px';
      if (window.matchMedia && window.matchMedia('(max-width: 480px)').matches) { fs = '12.5px'; ls = 0.5; pad = '17px 8px'; }
      if (window.matchMedia && window.matchMedia('(max-width: 340px)').matches) { fs = '11px'; ls = 0.2; pad = '16px 6px'; }
      btn.style.setProperty('font-size', fs, 'important');
      btn.style.setProperty('letter-spacing', ls + 'px', 'important');
      btn.style.setProperty('text-indent', ls + 'px', 'important');
      btn.style.setProperty('padding', pad, 'important');
      btn.style.setProperty('max-width', '100%', 'important');
    };

       inline (above). Braces: MEASURE the Name field and, if the button still
       differs, hand it the field's exact pixel width. Equal by construction,
       re-checked on every resize/rotation. */
    var equalize = function() {
      if (!btn || !sub) return;
      sub.style.setProperty('width', '100%', 'important');
      sub.style.setProperty('max-width', '100%', 'important');
      btn.style.setProperty('width', '100%', 'important');

      if (!btn) return;

      var cv = document.createElement('canvas').getContext('2d');

      var label = (btn.value || '').toUpperCase();

      for (; s >= 9; s -= 0.5) {
        cv.font = '600 ' + s + 'px Montserrat, sans-serif';
        if (cv.measureText(label).width + ls * label.length <= inner - 2) { break; }
      }
      btn.style.setProperty('font-size', s + 'px', 'important');
    };
    var applyBtn = function() { sizeBtn(); equalize(); fitLabel(); };

    window.setTimeout(applyBtn, 2600);

      window.addEventListener('load', applyBtn);

    if (oldParent && oldParent !== slot && oldParent.children.length === 0 && !(oldParent.textContent || '').replace(/\s+/g, '')) { oldParent.style.display = 'none'; }
  }, 200);
