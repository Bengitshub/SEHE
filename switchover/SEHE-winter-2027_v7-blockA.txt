<script>
(function(){
  var RAW = {
    'winter-2026'  : ['2026-06-27','2026-07-18','2026-08-01','2026-09-05','2026-09-12'],
    'pinnacle-2027': ['2027-01-14'],
    '11day-2627'   : ['2026-10-05','2026-11-02','2026-11-30','2027-02-01','2027-02-22','2027-03-01','2027-03-29'],
    '14day-2627'   : ['2026-10-12','2026-10-19','2026-10-26','2026-11-09','2026-11-16','2026-11-23','2027-01-04','2027-01-11','2027-01-18','2027-01-25','2027-02-08','2027-02-15','2027-03-08','2027-03-15','2027-03-22','2027-04-05','2027-04-12'],
    'winter-2027'  : ['2027-07-10','2027-07-24','2027-08-07','2027-08-14','2027-08-21','2027-08-28'],
    '11day-2728'   : ['2027-09-27','2027-10-25','2027-11-29','2028-01-10','2028-02-28','2028-03-06','2028-04-17'],
    '14day-2728'   : ['2027-10-04','2027-10-11','2027-10-18','2027-11-01','2027-11-08','2027-11-15','2027-11-22','2027-12-06','2028-01-17','2028-01-24','2028-01-31','2028-02-07','2028-02-14']
  };
  var SOLDOUT = {
    'winter-2026': ['2026-06-27','2026-07-18','2026-08-01'],
    '11day-2627' : ['2026-10-05','2026-11-02'],
    '14day-2627' : ['2026-10-19','2026-10-26']
  };
  var ADD = { 'winter-2026':11,'winter-2027':11,'pinnacle-2027':15,'11day-2627':10,'11day-2728':10,'14day-2627':13,'14day-2728':13 };
  function endOf(iso,n){ var d=new Date(iso+'T00:00:00Z'); d.setUTCDate(d.getUTCDate()+n); return d.toISOString().slice(0,10); }
  var today=new Date().toISOString().slice(0,10), data={};
  Object.keys(RAW).forEach(function(k){
    var sold=SOLDOUT[k]||[];
    var all=RAW[k].filter(function(x){return x>=today;}).map(function(x){return {start:x,end:endOf(x,ADD[k]),status:sold.indexOf(x)>=0?'soldout':'available'};});
    var av=all.filter(function(x){return x.status!=='soldout';});
    data[k]={ all:all, next: av.length?av[0].start:null };
  });
  window.__seheDepartureData=data;

  /* ---- LIVE DATES (the worker is the source of truth) --------------------
     The worker holds the full schedule and reads Checkfront for what's available
     vs sold out. We OVERRIDE the baked list per tour with the worker's live list,
     so sell-outs, freed cancellations and twin-share dates all update on their
     own — no re-pasting. SAFETY: if the worker is unreachable, or returns nothing
     for a tour, we keep the baked list (we never wipe it to empty). */
  var WORKER_URL='https://sehe-next-departures.ben-757.workers.dev/';
  window.__seheDeparturePromise=null;
  window.__seheLoadDepartures=function(){
    if(window.__seheDeparturePromise) return window.__seheDeparturePromise;
    window.__seheDeparturePromise=fetch(WORKER_URL,{cache:'no-store'})
      .then(function(r){ if(!r.ok) throw new Error('Worker '+r.status); return r.json(); })
      .then(function(live){
        if(live && typeof live==='object' && !Array.isArray(live)){
          Object.keys(window.__seheDepartureData).forEach(function(k){
            var lt=live[k];
            if(!lt || !lt.all || !lt.all.length) return;   /* nothing live for this tour -> keep the baked list (never wipe) */
            var all=lt.all.filter(function(d){ return d && d.start; }).map(function(d){
              return { start:d.start, end:d.end||endOf(d.start,ADD[k]), status:d.status||'available' };
            });
            if(!all.length) return;
            window.__seheDepartureData[k].all=all;                       /* worker wins */
            var av=all.filter(function(d){ return d.status!=='soldout'; });
            window.__seheDepartureData[k].next= lt.next || (av.length?av[0].start:null);
          });
        }
        return window.__seheDepartureData;
      })
      .catch(function(err){ if(window.console&&console.warn) console.warn('SEHE departures: live feed unavailable, keeping the in-page list.',err); return window.__seheDepartureData; });
    return window.__seheDeparturePromise;
  };
})();
</script>
