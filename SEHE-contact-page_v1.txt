<script>
(function () {
  /* Contact page enhancements — add only, never remove.
     1) "Start a chat" appears only once tawk.to (already on every page) is ready.
     2) Accessible names for the native form's three fields (its labels are
        payload keys and are NOT changed; aria-label only affects screen readers).
     No less-than sign and no ampersand in this script (Duda's publisher escapes both). */
  var tries = 0;
  var NAMES = { 'dmform-0': 'Your name', 'dmform-1': 'Your email address', 'dmform-3': 'Your message' };
  function chat() {
    var box = document.querySelector('.sehe-pg-contact [data-sehe-chat]');
    if (!box) return true;
    if (window.Tawk_API) {
      if (typeof window.Tawk_API.maximize === 'function') {
        box.hidden = false;
        var btn = box.querySelector('button');
        if (btn) {
          if (!btn.getAttribute('data-bound')) {
            btn.setAttribute('data-bound', '1');
            btn.addEventListener('click', function () { window.Tawk_API.maximize(); });
          }
        }
        return true;
      }
    }
    return false;
  }
  function names() {
    var details = document.querySelector('.sehe-pg-contact.ct-details');
    if (!details) return true;
    var row = details.closest('.dmRespRow');
    var form = row ? row.querySelector('.dmform') : null;
    if (!form) return false;
    Object.keys(NAMES).forEach(function (n) {
      var f = form.querySelector('[name="' + n + '"]');
      if (f) {
        f.setAttribute('aria-label', NAMES[n]);
        var wrap = f.closest('.dmforminput');
        if (wrap) { if (wrap.classList.contains('required')) f.setAttribute('aria-required', 'true'); }
      }
    });
    return true;
  }
  var doneChat = false, doneNames = false;
  var timer = setInterval(function () {
    tries = tries + 1;
    if (!doneChat) doneChat = chat();
    if (!doneNames) doneNames = names();
    if (doneChat) { if (doneNames) clearInterval(timer); }
    if (tries === 60) clearInterval(timer);
  }, 500);
})();
</script>
