  (function() {
  var days = document.querySelectorAll('.sehe-day');
  function updateExpandToggleState() {
    var expandBtn = document.getElementById('seheExpandAllBtn');
    var expandText = expandBtn ? expandBtn.querySelector('[data-expand-text]') : null;
    if (!expandBtn || !expandText) return;
    var openCount = document.querySelectorAll('.sehe-day.is-open').length;
    var allOpen = openCount === days.length;
    expandBtn.classList.toggle('is-expanded', allOpen);
    expandText.textContent = allOpen ? 'Collapse all' : 'Expand all';
    expandBtn.setAttribute('aria-pressed', allOpen);
  }
  days.forEach(function(day) {
    var btn = day.querySelector('.sehe-day-summary');
    if (!btn) return;
    btn.addEventListener('click', function() {
      var isOpen = day.classList.contains('is-open');
      day.classList.toggle('is-open', !isOpen);
      btn.setAttribute('aria-expanded', !isOpen);
      updateExpandToggleState();
    });
  });
  var expandBtn = document.getElementById('seheExpandAllBtn');
  if (expandBtn) {
    expandBtn.addEventListener('click', function() {
      var allOpen = document.querySelectorAll('.sehe-day.is-open').length === days.length;
      days.forEach(function(day) {
        var btn = day.querySelector('.sehe-day-summary');
        day.classList.toggle('is-open', !allOpen);
        if (btn) btn.setAttribute('aria-expanded', !allOpen);
      });
      updateExpandToggleState();
    });
  }
  var jumps = document.querySelectorAll('.sehe-day-jump');
  jumps.forEach(function(jump) {
    jump.addEventListener('click', function(e) {
      e.preventDefault();
      var dayNum = jump.getAttribute('data-jump');
      var target = document.getElementById('seheDay' + dayNum);
      if (!target) return;
      target.classList.add('is-open');
      var btn = target.querySelector('.sehe-day-summary');
      if (btn) btn.setAttribute('aria-expanded', 'true');
      updateExpandToggleState();
      var rect = target.getBoundingClientRect();
      var scrollTop = window.pageYOffset || document.documentElement.scrollTop;
      window.scrollTo({ top: rect.top + scrollTop - 40, behavior: 'smooth' });
    });
  });
})();

<script>
  (function() {
  var frame = document.getElementById('seheVideoFrame');
  var modal = document.getElementById('seheVideoModal');
  var content = document.getElementById('seheVideoModalContent');
  var closeBtn = document.getElementById('seheVideoClose');
  if (!frame || !modal || !content) return;
  function openModal() {
    var src = frame.getAttribute('data-video-src');
    var type = (frame.getAttribute('data-video-type') || 'mp4').toLowerCase();
    if (!src || src === 'REPLACE_WITH_VIDEO_URL_OR_ID') { console.warn('SEHE video: no video source set on .sehe-video-frame'); return; }
    var existing = content.querySelectorAll('video, iframe');
    existing.forEach(function(el) { el.remove(); });
    if (type === 'youtube') {
      var iframe = document.createElement('iframe');
      iframe.src = 'https://www.youtube.com/embed/' + src + '?autoplay=1&rel=0';
      iframe.allow = 'autoplay; encrypted-media; picture-in-picture; fullscreen';
      iframe.allowFullscreen = true;
      content.appendChild(iframe);
    } else if (type === 'vimeo') {
      var iframe = document.createElement('iframe');
      iframe.src = 'https://player.vimeo.com/video/' + src + '?autoplay=1';
      iframe.allow = 'autoplay; fullscreen; picture-in-picture';
      iframe.allowFullscreen = true;
      content.appendChild(iframe);
    } else {
      var video = document.createElement('video');
      video.src = src; video.controls = true; video.autoplay = true; video.playsInline = true;
      content.appendChild(video);
    }
    modal.classList.add('is-open');
    document.body.style.overflow = 'hidden';
  }
  function closeModal() {
    modal.classList.remove('is-open');
    document.body.style.overflow = '';
    var v = content.querySelector('video');
    if (v) { try { v.pause(); } catch(e){} }
    var existing = content.querySelectorAll('video, iframe');
    existing.forEach(function(el) { el.remove(); });
  }
  frame.addEventListener('click', openModal);
  frame.addEventListener('keydown', function(e) { if (e.key === 'Enter' || e.key === ' ') { e.preventDefault(); openModal(); } });
  closeBtn.addEventListener('click', closeModal);
  modal.addEventListener('click', function(e) { if (e.target === modal) closeModal(); });
  document.addEventListener('keydown', function(e) { if (e.key === 'Escape' && modal.classList.contains('is-open')) closeModal(); });
})();
</script>

  (function() {
  var slider = document.getElementById('seheGallerySlider');
  if (!slider) return;
  var track = document.getElementById('seheGalleryTrack');
  var slides = track.querySelectorAll('.sehe-gallery-slide');
  var dots = slider.querySelectorAll('.sehe-gallery-dot');
  var prevBtn = document.getElementById('seheGalleryPrev');
  var nextBtn = document.getElementById('seheGalleryNext');
  var counter = document.getElementById('seheGalleryCurrent');
  var total = slides.length;
  var current = 0;
  var autoTimer = null;
  var AUTO_DELAY = 5000;
  var paused = false;
  function pad2(n) { return n < 10 ? '0' + n : '' + n; }
  function goTo(index) {
    current = ((index % total) + total) % total;
    track.style.transform = 'translateX(-' + (current * 100) + '%)';
    dots.forEach(function(dot, i) { dot.classList.toggle('is-active', i === current); });
    if (counter) counter.textContent = pad2(current + 1);
  }
  function next() { goTo(current + 1); }
  function prev() { goTo(current - 1); }
  function startAuto() { stopAuto(); if (paused) return; autoTimer = setInterval(next, AUTO_DELAY); }
  function stopAuto() { if (autoTimer) { clearInterval(autoTimer); autoTimer = null; } }
  function resetAuto() { stopAuto(); startAuto(); }
  slider.addEventListener('mouseenter', function() { paused = true; stopAuto(); });
  slider.addEventListener('mouseleave', function() { paused = false; startAuto(); });
  nextBtn.addEventListener('click', function() { next(); resetAuto(); });
  prevBtn.addEventListener('click', function() { prev(); resetAuto(); });
  dots.forEach(function(dot) { dot.addEventListener('click', function() { var i = parseInt(dot.getAttribute('data-slide'), 10); if (!isNaN(i)) { goTo(i); resetAuto(); } }); });
  var touchStartX = 0; var touchStartY = 0; var touchEndX = 0; var touchEndY = 0; var touchActive = false;
  slider.addEventListener('touchstart', function(e) { touchStartX = e.touches[0].clientX; touchStartY = e.touches[0].clientY; touchActive = true; stopAuto(); }, { passive: true });
  slider.addEventListener('touchmove', function(e) { if (!touchActive) return; touchEndX = e.touches[0].clientX; touchEndY = e.touches[0].clientY; }, { passive: true });
  slider.addEventListener('touchend', function() {
    if (!touchActive) return;
    touchActive = false;
    var dx = touchStartX - touchEndX; var dy = touchStartY - touchEndY;
    if (Math.abs(dx) > 40 && Math.abs(dx) > Math.abs(dy)) { if (dx > 0) next(); else prev(); }
    startAuto();
  });
  slider.setAttribute('tabindex', '0');
  slider.addEventListener('keydown', function(e) { if (e.key === 'ArrowLeft') { prev(); resetAuto(); } else if (e.key === 'ArrowRight') { next(); resetAuto(); } });
  document.addEventListener('visibilitychange', function() { if (document.hidden) stopAuto(); else if (!paused) startAuto(); });
  startAuto();
})();
</script>

<!-- ===== HERO DEPARTURES — auto-fill the count + season range from the master list ===== -->
<script>
(function(){
  /* Which tour this page is. Change per tour when templating. */
  var SEHE_PAGE_TOUR = 'pinnacle-2027';
  var MS = ['Jan','Feb','Mar','Apr','May','Jun','Jul','Aug','Sep','Oct','Nov','Dec'];
  var MF = ['January','February','March','April','May','June','July','August','September','October','November','December'];
  function lbl(iso){ var p = iso.split('-'); return MS[parseInt(p[1],10)-1] + ' ' + p[0].slice(2); }
  function full(iso){ var p = iso.split('-'); return parseInt(p[2],10) + ' ' + MF[parseInt(p[1],10)-1] + ' ' + p[0]; }
  function paint(){
    var d = (window.__seheDepartureData || {})[SEHE_PAGE_TOUR];
    if (!d || !d.all || !d.all.length) return;
    var avail = d.all.filter(function(x){ return x && x.status !== 'soldout'; });
    var list = avail.length ? avail : d.all;
    var cEl = document.querySelector('[data-sehe-dep-count]');
    if (cEl) cEl.textContent = list.length;
    var rEl = document.querySelector('[data-sehe-dep-range]');
    if (rEl) rEl.innerHTML = (list.length === 1) ? full(list[0].start) : (lbl(list[0].start) + ' – ' + lbl(list[list.length - 1].start));
  }
  paint();   /* instant paint from fallback / already-loaded live data */
  if (typeof window.__seheLoadDepartures === 'function') { window.__seheLoadDepartures().then(paint); }   /* re-render with live */
})();
</script>
